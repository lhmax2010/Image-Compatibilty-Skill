"""Independent read-only audit of previous phases; outputs only in phase5/work."""
import collections, hashlib, json, os, pathlib, stat, sys, time
B=pathlib.Path('<ACCEPTANCE_ROOT>');W=B/'work/phase5'
def sha(p):
 h=hashlib.sha256()
 with p.open('rb') as f:
  for block in iter(lambda:f.read(1048576),b''):h.update(block)
 return h.hexdigest()
def reports():
 result={}
 for phase in ('phase3-quick','phase4-static'):
  for p in sorted((B/'reports'/phase).rglob('*')):
   if p.is_file():result[str(p.relative_to(B))]={'size':p.stat().st_size,'sha256':sha(p)}
 return result
mode=sys.argv[1];started=time.monotonic()
if mode=='before':
 data=reports();(W/'REPORTS_BEFORE.json').write_text(json.dumps(data,indent=2));print('snapshotted',len(data),'files',time.monotonic()-started,flush=True)
else:
 errors=[];data=reports();before=json.loads((W/'REPORTS_BEFORE.json').read_text());out={'previous_reports_unchanged':data==before,'previous_report_files':len(data),'rootfs':{},'archives':[],'partitions':[]}
 if data!=before:errors.append('Previous reports changed')
 for side in ('baseline','candidate'):
  root=B/'rootfs'/side;snap=json.loads((B/'reports/phase2-extraction'/f'{side}-output-snapshot.json').read_text());actual={''};bytes_read=0
  for base,dirs,files in os.walk(root,followlinks=False):
   for name in dirs+files:actual.add(str((pathlib.Path(base)/name).relative_to(root)))
  if actual!={x['path'] for x in snap}:errors.append(side+' rootfs path set changed')
  for x in snap:
   p=root/x['path'];s=p.lstat()
   if oct(stat.S_IMODE(s.st_mode))!=x['output_mode'] or s.st_uid!=x['output_uid'] or s.st_gid!=x['output_gid']:errors.append([side,x['path'],'metadata'])
   if x['type']=='regular':
    if not stat.S_ISREG(s.st_mode) or s.st_size!=x['size'] or sha(p)!=x['sha256']:errors.append([side,x['path'],'content'])
    bytes_read+=s.st_size
   elif x['type']=='symlink':
    if not stat.S_ISLNK(s.st_mode) or os.readlink(p)!=x['symlink_target']:errors.append([side,x['path'],'symlink'])
   elif not stat.S_ISDIR(s.st_mode):errors.append([side,x['path'],'directory'])
  out['rootfs'][side]={'entries':len(snap),'regular_bytes_hashed':bytes_read}
  print('rootfs checked',side,flush=True)
 for x in json.loads((B/'reports/phase1-download/DOWNLOAD_MANIFEST.json').read_text())['items']:
  same=sha(pathlib.Path(x['path']))==x['actual_sha256'];out['archives'].append({'path':x['path'],'unchanged':same})
  if not same:errors.append('archive changed '+x['path'])
 for x in json.loads((B/'reports/phase2-extraction/PARTITION_IMMUTABILITY_FINAL.json').read_text())['partitions']:
  same=sha(pathlib.Path(x['path']))==x['sha256_after_fs_export'];out['partitions'].append({'path':x['path'],'unchanged':same})
  if not same:errors.append('partition changed '+x['path'])
 out.update(status='FAIL' if errors else 'PASS',errors=errors,wall_seconds=time.monotonic()-started)
 (W/'INPUT_IMMUTABILITY_AFTER.json').write_text(json.dumps(out,indent=2));print(json.dumps(out),flush=True)
 sys.exit(bool(errors))
